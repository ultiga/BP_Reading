  <script>

    let uploadedFileUrl = '';


    function handleFile(input) {

      if (!input.files || !input.files[0]) return;

      const file = input.files[0];


      // Preview Image

      const reader = new FileReader();

      reader.onload = function(e) {

        document.getElementById('imagePreview').src = e.target.result;

        document.getElementById('previewBox').style.display = 'block';


        showStatus('Uploading & scanning BP monitor screen with OCR...', 'status-loading');

        document.getElementById('submitBtn').disabled = true;


        google.script.run

          .withSuccessHandler(onOcrSuccess)

          .withFailureHandler(onOcrFailure)

          .processUploadedImage(e.target.result, file.name, file.type);

      };

      reader.readAsDataURL(file);

    }


    function onOcrSuccess(res) {

      if (!res.success) {

        showStatus('Error processing image: ' + res.error, 'status-error');

        return;

      }


      uploadedFileUrl = res.fileUrl;

      document.getElementById('fileUrl').value = uploadedFileUrl;


      // Populate parsed fields

      if (res.extracted.systolic) document.getElementById('systolic').value = res.extracted.systolic;

      if (res.extracted.diastolic) document.getElementById('diastolic').value = res.extracted.diastolic;

      if (res.extracted.pulse) document.getElementById('pulse').value = res.extracted.pulse;


      showStatus('Image processed! Verify or adjust readings, then save.', 'status-success');

      document.getElementById('submitBtn').disabled = false;

    }


    function onOcrFailure(err) {

      showStatus('Failed to connect to Google Apps Script: ' + err, 'status-error');

    }


    function handleFormSubmit(e) {

      e.preventDefault();

      const submitBtn = document.getElementById('submitBtn');

      submitBtn.disabled = true;

      showStatus('Submitting record to Google Sheet...', 'status-loading');


      const payload = {

        systolic: document.getElementById('systolic').value,

        diastolic: document.getElementById('diastolic').value,

        pulse: document.getElementById('pulse').value,

        fileUrl: uploadedFileUrl

      };


      google.script.run

        .withSuccessHandler(function(res) {

          if (res.success) {

            showStatus(`Successfully saved! Record #${res.nextNo} created.`, 'status-success');

            document.getElementById('bpForm').reset();

            document.getElementById('previewBox').style.display = 'none';

          } else {

            showStatus('Error saving data: ' + res.error, 'status-error');

            submitBtn.disabled = false;

          }

        })

        .withFailureHandler(function(err) {

          showStatus('Error: ' + err, 'status-error');

          submitBtn.disabled = false;

        })

        .submitBPRecord(payload);

    }


    function showStatus(msg, className) {

      const alert = document.getElementById('statusAlert');

      alert.className = 'status-badge ' + className;

      alert.textContent = msg;

    }
